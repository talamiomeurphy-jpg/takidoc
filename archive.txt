    <script type="module">
        import { templatesRegistry } from './js/templates-registry.js';

        const SUPABASE_URL = 'https://wyfkogowsdbxctbpfuud.supabase.co';
        const SUPABASE_ANON_KEY = 'eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJpc3MiOiJzdXBhYmFzZSIsInJlZiI6Ind5ZmtvZ293c2RieGN0YnBmdXVkIiwicm9sZSI6ImFub24iLCJpYXQiOjE3OTA3MDc3MjAsImV4cCI6MjEwNjI4MzcyMH0.231eQ38RFvFaH3O6D-ReA8rlf3TehvWfcM-LBWocNEM';
        
        const supabase = window.supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

        const grid = document.getElementById('masonryGrid');
        const searchInput = document.getElementById('searchInput');
        const catPills = document.querySelectorAll('.cat-pill');
        const notifDropdown = document.getElementById('notifDropdown');
        const headerActions = document.getElementById('headerActions');
        const navSearch = document.getElementById('nav-search');

        let currentUser = null;
        let thumbnailsCache = {};

        // ============================================
        // AUTHENTIFICATION
        // ============================================
        async function checkAuth() {
            try {
                const { data: { session } } = await supabase.auth.getSession();
                currentUser = session?.user || null;
                updateHeaderUI();
            } catch (error) {
                console.error('Erreur auth:', error);
            }
        }

        function updateHeaderUI() {
            const notifBtnHTML = `<button class="icon-btn" id="notifBtn"><i class="far fa-bell"></i><span class="badge">3</span></button>`;
            
            if (currentUser) {
                const fullName = currentUser.user_metadata?.full_name || currentUser.email || 'Utilisateur';
                const initials = fullName.split('@')[0].split(' ').map(n => n[0]).join('').toUpperCase().slice(0, 2);
                headerActions.innerHTML = notifBtnHTML + `<a href="profile.html" class="btn-profile"><span class="initials">${initials}</span><span>Profil</span></a>`;
            } else {
                headerActions.innerHTML = notifBtnHTML + `<a href="auth.html?action=signup" class="btn-auth btn-signup"><i class="fas fa-user-plus"></i><span>S'inscrire</span></a><a href="auth.html?action=login" class="btn-auth btn-login"><i class="fas fa-sign-in-alt"></i><span>Connexion</span></a>`;
            }
            
            const notifBtn = document.getElementById('notifBtn');
            if (notifBtn) {
                notifBtn.addEventListener('click', (e) => {
                    e.stopPropagation();
                    notifDropdown.classList.toggle('active');
                });
            }
        }

        const categoryNames = { 'cv': 'CV', 'lettre': 'Lettre', 'contrat': 'Contrat', 'commercial': 'Commercial', 'attestation': 'Attestation', 'administratif': 'Administratif' };
        const categoryIcons = { 'cv': 'fa-user-tie', 'lettre': 'fa-envelope-open-text', 'contrat': 'fa-file-signature', 'commercial': 'fa-file-invoice-dollar', 'attestation': 'fa-certificate', 'administratif': 'fa-building' };

        document.addEventListener('click', () => notifDropdown.classList.remove('active'));

        // ============================================
        // PLACEHOLDERS STYLISÉS PAR CATÉGORIE (Affichage immédiat)
        // ============================================
        function getPlaceholderHTML(template) {
            const icon = categoryIcons[template.category] || 'fa-file-alt';
            const name = template.name.toUpperCase();
            return `
                <div class="thumbnail-placeholder">
                    <div class="placeholder-icon"><i class="fas ${icon}"></i></div>
                    <div class="placeholder-title">${name}</div>
                    <div class="placeholder-lines">
                        <div class="placeholder-line medium"></div>
                        <div class="placeholder-line"></div>
                        <div class="placeholder-line short"></div>
                        <div class="placeholder-line medium"></div>
                        <div class="placeholder-line"></div>
                    </div>
                </div>
            `;
        }

        // ============================================
        // GÉNÉRATION LAZY DES MINIATURES (En arrière-plan)
        // ============================================
        async function generateThumbnail(template) {
            // Vérifier le cache d'abord
            const cached = localStorage.getItem(`thumb_${template.id}`);
            if (cached) return cached;

            try {
                // Créer un conteneur hors écran
                const container = document.createElement('div');
                container.style.cssText = 'position: absolute; left: -10000px; top: 0; width: 794px; background: white; font-family: inherit;';
                
                // Injecter le CSS du template
                const style = document.createElement('style');
                style.textContent = template.cssStyles || '';
                container.appendChild(style);
                
                // Injecter le HTML
                const content = document.createElement('div');
                content.innerHTML = template.htmlStructure || '<p>Document</p>';
                container.appendChild(content);
                
                document.body.appendChild(container);
                
                // Attendre le rendu
                await new Promise(resolve => setTimeout(resolve, 200));
                
                // Capturer avec html2canvas (hauteur fixe A4)
                const canvas = await html2canvas(container, {
                    scale: 0.4,
                    useCORS: true,
                    logging: false,
                    backgroundColor: '#ffffff',
                    width: 794,
                    height: 1123,
                    windowWidth: 794,
                    windowHeight: 1123
                });
                
                const imageData = canvas.toDataURL('image/jpeg', 0.85);
                
                // Mettre en cache
                try {
                    localStorage.setItem(`thumb_${template.id}`, imageData);
                } catch (e) {
                    console.warn('Cache plein:', e);
                }
                
                // Nettoyer
                document.body.removeChild(container);
                
                return imageData;
            } catch (error) {
                console.error(`Erreur miniature ${template.id}:`, error);
                // Nettoyer en cas d'erreur
                const existing = document.body.querySelector(`[style*="left: -10000px"]`);
                if (existing) document.body.removeChild(existing);
                return null;
            }
        }

        // ============================================
        // AFFICHAGE DES CARTES (Avec mise à jour progressive)
        // ============================================
        function renderPins(docs) {
            grid.innerHTML = '';
            
            if (docs.length === 0) {
                grid.innerHTML = `<div class="empty-state"><i class="fas fa-search"></i><h3>Aucun document trouvé</h3><p>Essayez une autre recherche</p></div>`;
                return;
            }

            docs.forEach(template => {
                const card = document.createElement('div');
                card.className = 'pin-card';
                card.dataset.templateId = template.id;
                
                const categoryName = categoryNames[template.category] || template.category;
                const isSaved = localStorage.getItem('takidoc_favorites')?.includes(template.id);
                const cachedThumb = thumbnailsCache[template.id];
                
                card.innerHTML = `
                    <div class="document-preview">
                        <div class="category-badge">${categoryName}</div>
                        ${cachedThumb 
                            ? `<img src="${cachedThumb}" alt="${template.name}" class="thumbnail-image" loading="lazy">`
                            : getPlaceholderHTML(template)
                        }
                        <div class="pin-overlay">
                            <div class="overlay-top">
                                <button class="overlay-btn save ${isSaved ? 'saved' : ''}" onclick="event.stopPropagation(); toggleSave('${template.id}', this)"><i class="${isSaved ? 'fas' : 'far'} fa-heart"></i></button>
                                <button class="overlay-btn" onclick="event.stopPropagation(); shareDoc('${template.id}', '${template.name.replace(/'/g, "\\'")}')" title="Partager"><i class="fas fa-share"></i></button>
                            </div>
                            <div class="overlay-bottom">
                                <h3 class="overlay-title">${template.name}</h3>
                                <div class="overlay-price">${template.price.toLocaleString()} FCFA</div>
                                <button class="btn-customize-overlay" onclick="event.stopPropagation(); window.location.href='editor.html?id=${template.id}'"><i class="fas fa-edit"></i> Personnaliser</button>
                            </div>
                        </div>
                    </div>
                `;
                
                card.addEventListener('click', () => window.location.href = `editor.html?id=${template.id}`);
                grid.appendChild(card);
            });

            // Générer les miniatures en arrière-plan pour celles qui n'ont pas de cache
            generateThumbnailsInBackground(docs);
        }

        async function generateThumbnailsInBackground(docs) {
            for (const template of docs) {
                if (thumbnailsCache[template.id]) continue; // Déjà en cache
                
                const thumbnail = await generateThumbnail(template);
                if (thumbnail) {
                    thumbnailsCache[template.id] = thumbnail;
                    
                    // Mettre à jour la carte si elle est encore visible
                    const card = document.querySelector(`.pin-card[data-template-id="${template.id}"]`);
                    if (card) {
                        const preview = card.querySelector('.document-preview');
                        const placeholder = preview.querySelector('.thumbnail-placeholder');
                        if (placeholder) {
                            const img = document.createElement('img');
                            img.src = thumbnail;
                            img.alt = template.name;
                            img.className = 'thumbnail-image';
                            img.style.opacity = '0';
                            img.style.transition = 'opacity 0.5s ease';
                            placeholder.replaceWith(img);
                            // Fade in
                            setTimeout(() => { img.style.opacity = '1'; }, 50);
                        }
                    }
                }
            }
        }

        // ============================================
        // FILTRAGE ET RECHERCHE
        // ============================================
        let currentCat = 'all';
        catPills.forEach(pill => {
            pill.addEventListener('click', () => {
                catPills.forEach(p => p.classList.remove('active'));
                pill.classList.add('active');
                currentCat = pill.dataset.cat;
                filterAndRender();
            });
        });

        searchInput.addEventListener('input', filterAndRender);

        function filterAndRender() {
            const query = searchInput.value.toLowerCase().trim();
            const filtered = templatesRegistry.filter(doc => {
                const matchCat = currentCat === 'all' || doc.category === currentCat;
                const matchSearch = !query || doc.name.toLowerCase().includes(query) || doc.description.toLowerCase().includes(query);
                return matchCat && matchSearch;
            });
            renderPins(filtered);
        }

        // ============================================
        // ACTIONS GLOBALES
        // ============================================
        window.toggleSave = function(docId, btn) {
            const icon = btn.querySelector('i');
            let favorites = JSON.parse(localStorage.getItem('takidoc_favorites') || '[]');
            if (btn.classList.contains('saved')) {
                btn.classList.remove('saved');
                icon.classList.remove('fas');
                icon.classList.add('far');
                favorites = favorites.filter(id => id !== docId);
            } else {
                btn.classList.add('saved');
                icon.classList.remove('far');
                icon.classList.add('fas');
                if (!favorites.includes(docId)) favorites.push(docId);
                btn.style.transform = 'scale(1.3)';
                setTimeout(() => btn.style.transform = '', 200);
            }
            localStorage.setItem('takidoc_favorites', JSON.stringify(favorites));
        };

        window.shareDoc = function(id, name) {
            const url = window.location.origin + '/editor.html?id=' + id;
            if (navigator.share) {
                navigator.share({ title: name + ' - TakiDoc', url }).catch(() => {});
            } else {
                navigator.clipboard.writeText(url).then(() => alert('Lien copié !'));
            }
        };

        window.markAllRead = function(e) {
            e.stopPropagation();
            document.querySelectorAll('.notif-item.unread').forEach(item => item.classList.remove('unread'));
        };

        // Navigation recherche
        if (navSearch) {
            navSearch.addEventListener('click', (e) => {
                e.preventDefault();
                window.scrollTo({ top: 0, behavior: 'smooth' });
                setTimeout(() => searchInput.focus(), 300);
            });
        }

        // ============================================
        // INITIALISATION
        // ============================================
        document.addEventListener('DOMContentLoaded', async () => {
            await checkAuth();
            renderPins(templatesRegistry);
        });
